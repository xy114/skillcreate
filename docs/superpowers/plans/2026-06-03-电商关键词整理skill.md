### Task 6: 构建分类引擎规则（Python 验证脚本）

**Files:**
- Create: `d:/skillcreate/skill/分类引擎.py`

> **为什么有这个文件？** 悟空工作流中的条件分支逻辑需要先在 Python 中验证正确性。此脚本**不是**最终在悟空上运行的代码，而是：
> - 用来验证分类规则是否产生正确结果
> - 作为悟空工作流配置节点的逻辑参考
> - 后续可以独立运行做回归测试

- [ ] **Step 1: 编写分类引擎脚本**

```python
#!/usr/bin/env python3
"""
关键词分类引擎 —— 悟空工作流核心逻辑的 Python 验证版本

用于验证分类规则的正确性，并作为悟空工作流节点配置的逻辑参考。
悟空平台的工作流节点将基于此脚本中定义的规则进行配置。
"""

import json
import csv
import sys
from pathlib import Path

# ============================================================
# 加载词库
# ============================================================

SKILL_DIR = Path(__file__).parent
词库_DIR = SKILL_DIR / "词库"

def 加载_json(文件名: str) -> dict:
    """加载词库 JSON 文件"""
    with open(词库_DIR / 文件名, encoding="utf-8") as f:
        return json.load(f)

品牌竞品词库 = 加载_json("品牌竞品词库.json")
属性词库 = 加载_json("属性词库.json")
品类词库 = 加载_json("品类词库.json")
场景信号词库 = 加载_json("场景信号词库.json")

# 展平词库为集合，方便快速匹配
品牌词集合 = set(w.lower() for w in 品牌竞品词库["品牌词"])
竞品词集合 = set(w.lower() for w in 品牌竞品词库["竞品词"])
品类词集合 = set(w.lower() for w in 品类词库["品类词列表"])

# 属性词按维度分别展平
功能词集合 = set(w.lower() for w in 属性词库["属性维度"]["功能属性"]["词列表"])
规格词集合 = set(w.lower() for w in 属性词库["属性维度"]["规格属性"]["词列表"])
人群词集合 = set(w.lower() for w in 属性词库["属性维度"]["人群属性"]["词列表"])
场景属性集合 = set(w.lower() for w in 属性词库["属性维度"]["场景属性"]["词列表"])
所有属性词集合 = 功能词集合 | 规格词集合 | 人群词集合 | 场景属性集合

# 场景信号词展平
痛点信号词集合 = set(w.lower() for w in 场景信号词库["信号词"]["痛点解决型"]["词列表"])
场景适配词集合 = set(w.lower() for w in 场景信号词库["信号词"]["场景适配型"]["词列表"])
时效需求词集合 = set(w.lower() for w in 场景信号词库["信号词"]["时效需求型"]["词列表"])
所有信号词集合 = 痛点信号词集合 | 场景适配词集合 | 时效需求词集合

# ============================================================
# 分类引擎
# ============================================================

def 提取关键词中的所有单词(关键词: str) -> set:
    """将关键词按空格拆分为单词集合，支持中英文混合"""
    return set(关键词.lower().split())

def 分类关键词(关键词: str) -> tuple:
    """
    对单个关键词进行分类
    返回: (分类结果, 属性标签列表, 命中信号词列表)
    """
    单词 = 提取关键词中的所有单词(关键词)

    # ---- 第 1 步: 品牌/竞品词匹配 ----
    命中品牌 = 单词 & 品牌词集合
    if 命中品牌:
        return ("品牌词", [], [])
    
    命中竞品 = [w for w in 单词 if any(cw in w for cw in 竞品词集合)]
    if 命中竞品:
        return ("竞品词", [], [])

    # ---- 第 2 步: 属性词匹配 ----
    命中维度 = []
    属性标签 = []
    命中的属性词 = []

    if 单词 & 功能词集合:
        命中维度.append("功能属性")
        属性标签.append(f"功能:{','.join(单词 & 功能词集合)}")
        命中的属性词.extend(单词 & 功能词集合)
    
    if 单词 & 规格词集合:
        命中维度.append("规格属性")
        属性标签.append(f"规格:{','.join(单词 & 规格词集合)}")
        命中的属性词.extend(单词 & 规格词集合)
    
    if 单词 & 人群词集合:
        命中维度.append("人群属性")
        属性标签.append(f"人群:{','.join(单词 & 人群词集合)}")
        命中的属性词.extend(单词 & 人群词集合)

    if 单词 & 场景属性集合:
        命中维度.append("场景属性")
        属性标签.append(f"场景:{','.join(单词 & 场景属性集合)}")
        命中的属性词.extend(单词 & 场景属性集合)

    命中信号词 = 单词 & 所有信号词集合

    # 判定分类
    维度数 = len(命中维度)

    # ≥2 个属性维度 → 二级属性词
    if 维度数 >= 2:
        return ("二级属性词", 属性标签, list(命中信号词))

    # 1 个属性维度 + 场景/痛点信号词 → 三级场景词
    if 维度数 == 1 and 命中信号词:
        return ("三级场景词", 属性标签, list(命中信号词))

    # 仅 1 个属性维度 → 二级属性词
    if 维度数 == 1:
        return ("二级属性词", 属性标签, list(命中信号词))

    # 仅命中信号词，无属性词 → 三级场景词
    if 命中信号词:
        return ("三级场景词", [], list(命中信号词))

    # ---- 第 3 步: 纯品类词判定 ----
    命中品类 = 单词 & 品类词集合
    if 命中品类:
        return ("一级核心词", [], [])

    # ---- 第 4 步: 相关性兜底 ----
    # 如果关键词与类目有部分重叠（包含类目词片段），保留为不相关待人工审核
    # 否则标记为不相关/宽泛词
    return ("不相关/宽泛词", [], [])


def 计算建议动作(分类: str, 词频: int, 搜索量_str: str) -> str:
    """基于分类结果和指标建议投放策略"""
    if 分类 in ("品牌词", "竞品词"):
        return "建议投放"
    if 分类 == "不相关/宽泛词":
        return "建议否定"
    if 分类 == "一级核心词":
        return "仅Listing嵌入"
    # 二级、三级：投放主力
    if 分类 in ("二级属性词", "三级场景词"):
        return "建议投放"
    return "待判定"


def 处理关键词表(输入_csv路径: str, 输出_csv路径: str) -> dict:
    """读取原始 CSV，逐词分类，输出结构化结果 CSV，返回统计摘要"""
    
    # 去重计数
    seen = {}
    rows = []
    
    with open(输入_csv路径, encoding="utf-8") as f:
        reader = csv.DictReader(f)
        raw_rows = list(reader)
    
    for row in raw_rows:
        kw = row["关键词"].strip().lower()
        if kw not in seen:
            seen[kw] = {
                "关键词": row["关键词"].strip(),
                "搜索量": row.get("搜索量", ""),
                "竞争度": row.get("竞争度", ""),
                "PPC竞价(USD)": row.get("PPC竞价(USD)", ""),
                "词频": 1
            }
        else:
            seen[kw]["词频"] += 1
        rows.append(row)
    
    # 逐词分类
    结果 = []
    统计 = {
        "总词数": len(rows),
        "去重后": len(seen),
        "一级核心词": 0,
        "二级属性词": 0,
        "三级场景词": 0,
        "品牌词": 0,
        "竞品词": 0,
        "不相关/宽泛词": 0,
        建议否定词: 0,
    }
    
    for kw, info in seen.items():
        分类, 属性标签, 信号词 = 分类关键词(kw)
        建议动作 = 计算建议动作(分类, info["词频"], info["搜索量"])
        
        # 统计
        if 分类 == "一级核心词":
            统计["一级核心词"] += 1
        elif 分类 == "二级属性词":
            统计["二级属性词"] += 1
        elif 分类 == "三级场景词":
            统计["三级场景词"] += 1
        elif 分类 == "品牌词":
            统计["品牌词"] += 1
        elif 分类 == "竞品词":
            统计["竞品词"] += 1
        elif 分类 == "不相关/宽泛词":
            统计["不相关/宽泛词"] += 1
            if 建议动作 == "建议否定":
                统计["建议否定词"] += 1
        
        结果.append({
            "关键词": info["关键词"],
            "分类": 分类,
            "属性标签": "; ".join(属性标签) if 属性标签 else "-",
            "词频": str(info["词频"]),
            "建议动作": 建议动作,
            "搜索量": info["搜索量"],
            "竞争度": info["竞争度"],
            "PPC竞价(USD)": info["PPC竞价(USD)"],
        })
    
    # 写入输出 CSV
    输出列 = ["关键词", "分类", "属性标签", "词频", "建议动作", "搜索量", "竞争度", "PPC竞价(USD)"]
    with open(输出_csv路径, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=输出列)
        writer.writeheader()
        writer.writerows(结果)
    
    return 统计


def 生成群消息摘要(统计: dict, 类目名: str) -> str:
    """根据统计结果生成钉钉群消息摘要"""
```

```javascript
// 对「关键词」列执行清洗

// 3. 去除多余空格（多个空格合并为一个）

function cleanKeyword(kw) {
    return kw.trim().toLowerCase().replace(/[^a-z0-9一-龥\s]/g, '').replace(/\s+/g, ' ');
```

### 步骤 2.2: 去重 + 词频统计
```javascript
// 按「关键词」列分组，计算每个唯一词的出现次数
// 新增列: 「词频」
// 输出: 去重后的表（每个关键词一行，词频列 = 原始出现次数）
function deduplicateAndCount(table, keywordCol) {
    const countMap = {};
    for (const row of table) {
        const kw = row[keywordCol];
        countMap[kw] = (countMap[kw] || 0) + 1;
    }
    const deduped = [];
    const seen = new Set();
    for (const row of table) {
        const kw = row[keywordCol];
        if (!seen.has(kw)) {
            seen.add(kw);
            deduped.push({...row, 词频: countMap[kw]});
        }
    }
    return deduped;
}
// 输出变量名: preprocessed_table
```
